```
SELECT 
    qURL.ID,
    qURL.Converted,
    q.CampaignID,
    q.Date,
    q.AccPackID,
    q.BrokenReason,
    q.BrokenAuto,
    q.BrokenInfo,
    JSON_EXTRACT(qURL.Info, '$.pages_info[0].size_page[0]') AS size,
    JSON_EXTRACT(qURL.Info, '$.pages_info[0].frm_page[0]') AS frm,
    JSON_EXTRACT(qURL.Info, '$.pages_info[0].dpi[0]') AS dpi
FROM 
    ClientClusterSBRNK.QuestionaryScanUrls qURL
JOIN 
    ClientClusterSBRNK.Questionaries q
ON 
    qURL.ID = q.ID
```
